```kotlin
@Composable
private fun TimeField(
    label: String,
    hour: Int?,
    minute: Int?,
    onPick: (Int) -> Unit
) {
    val initial = if (hour == null) "" else
        "%02d:%s".format(hour, minute?.let { "%02d".format(it) } ?: "00")
    var text by remember(hour, minute) { mutableStateOf(initial) }
    var invalid by remember(hour, minute) { mutableStateOf(false) }

    Column {
        Text(label, style = MaterialTheme.typography.labelSmall, color = TextMuted)
        OutlinedTextField(
            value = text,
            onValueChange = { raw ->
                // Chi nhan so va dau hai cham, toi da 5 ky tu "HH:MM".
                val cleaned = raw.filter { it.isDigit() || it == ':' }.take(5)
                text = cleaned
                val m = Regex("^(\\d{1,2}):(\\d{1,2})$").find(cleaned)
                if (m == null) {
                    invalid = cleaned.isNotEmpty()
                } else {
                    val h = m.groupValues[1].toInt()
                    val mi = m.groupValues[2].toInt()
                    if (h in 0..23 && mi in 0..59) {
                        invalid = false
                        onPick(h * 60 + mi)
                    } else {
                        invalid = true
                    }
                }
            },
            modifier = Modifier
                .fillMaxWidth()
                .testTag("block_window_${label.lowercase().filter { it.isLetter() }}"),
            singleLine = true,
            placeholder = { Text("21:00", color = TextMuted, fontSize = 14.sp) },
            isError = invalid,
            supportingText = if (invalid) {
                { Text("Phải là giờ thật, kiểu 21:00 hoặc 06:30", color = NeonRose, fontSize = 11.sp) }
            } else null,
            shape = RoundedCornerShape(16.dp),
            colors = OutlinedTextFieldDefaults.colors(
                focusedBorderColor = NeonCyan,
                unfocusedBorderColor = SpaceCardBorder,
                focusedTextColor = TextPrimary,
                unfocusedTextColor = TextPrimary
            )
        )
    }
}
```
